     (interactive)
     (org-agenda nil ,key)))

  "Create Org-capture shortcut functions for KEY."
  `(defun ,(intern (concat "+org-capture-key-" key)) ()
     (interactive)
     (org-capture nil ,key)))

  "Create Org-capture FILE shortcut functions for DESCRIPTION."
  `(defun ,(intern (concat "+org-capture-open-" description)) ()
     (interactive)
